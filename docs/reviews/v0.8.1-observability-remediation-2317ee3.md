@pytest.fixture

def write(path, records):

def start(**extra):
    return {'run_id':'r','mode':'search','status':'incomplete','ok':False,'applied':0,'funnel':{},**extra}

def event(**extra):
    return {'event':'application_result','outcome':'sent','run_id':'r','source':'hh','vacancy_id':'1','application_id':'a',**extra}

    return a

@pytest.mark.parametrize('loss',['lost','truncated'])

        with Path(config.RUN_HISTORY_FILE).open('a') as f:f.write('{"status":"finished",\n')

        assert 'Отклики: 0' not in text and 'Откликов: 0' not in text
